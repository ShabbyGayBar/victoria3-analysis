    Scenario dataclass ¶

    Scenario(
    objective: str = "automation",
    min_goods_balance: float | None = 0.0,
    banned_pms: tuple[str, ...] = (),
    building_level_caps: tuple[tuple[str, float], ...] = (),
    banned_building_groups: tuple[str, ...] = (),
    throughput_multipliers: tuple[
        tuple[str, float], ...
    ] = (),
    era_cap: int | None = None,
    construction_cost_cap: float | None = None,
    employment_cap: float | None = None,
    arable_land_cap: float | None = None,
    min_net_infrastructure: float | None = None,
    urbanization_per_center: float | None = None,
    name: str | None = None,
    goods_import: tuple[tuple[str, float], ...] = (),
    goods_export: tuple[tuple[str, float], ...] = (),
    workforce_ratio: float = 1.0,
)

    An immutable optimization recipe expressed as data.

    The optimizer interpreting a scenario owns all economy-dependent compilation. This object records assumptions interpreted by the nominal, market, and equilibrium formulations.

    min_goods_balance sets the minimum weekly balance for every good: building output plus imports minus building inputs and exports. Household consumption is excluded. The default of 0.0 requires nonnegative balances; negative floors allow deficits, and None disables the floor.

    min_net_infrastructure sets minimum infrastructure generation minus consumption, weighted by building levels. 0.0 prevents a deficit; positive minima require spare capacity, negative minima allow a bounded deficit, and None leaves infrastructure unconstrained.

    goods_import and goods_export record fixed weekly (good, amount) pairs with non-negative amounts. Imports add sell orders; exports add buy orders. Repeated entries are summed, and a good may be both imported and exported. These orders enter the goods-balance constraint when enabled.

    workforce_ratio is the uniform worker share of total population. It defaults to 1.0; employment constraints still count only workers.

    def display_name(self) -> str:

    def summary(self) -> pd.DataFrame:
    """Return a fresh one-row report of the latest returned solution.

    GDP is reported weekly and annually; ``gdp_per_capita`` is annual.
    Employment counts workers, population includes dependents, and wealth
    is a population-weighted tier. Infrastructure usage is consumption
    minus generation. Construction cost and arable-land consumption use
    the model's existing accounting. ``throughput_multiplier`` is NaN
    unless all compiled configuration multipliers are equal.

    Reporting never solves or evaluates the problem again. Binding a new
    problem clears the retained solution. Equilibrium searches that return
    a verified state on budget exhaustion remain reportable even when
    ``search_success`` is false.

    Raises:
        ValueError: If the current problem has no returned solution.
    """
    state = self._require_state()
    problem = self._require_problem()
    result = self.result
    if result is None:
        raise ValueError("Optimizer has no returned solution to summarize.")
    scenario = problem.scenario
    multipliers = problem.throughput_multipliers
    values: dict[str, object] = {
        "objective": scenario.objective,
        "employment_cap": scenario.employment_cap,
        "workforce_ratio": scenario.workforce_ratio,
        "banned_building_groups": ";".join(scenario.banned_building_groups),
        "min_net_infrastructure": scenario.min_net_infrastructure,
        "min_goods_balance": scenario.min_goods_balance,
        "configuration_count": len(self.model.building_index),
        "throughput_multiplier": (
            float(multipliers[0])
            if multipliers.size and np.all(multipliers == multipliers[0])
            else float("nan")
        ),
        "economy_of_scale_level_cap": 0.0,
        "gdp_weekly": state.gdp_weekly,
        "annual_gdp": state.gdp(annual=True),
        "employment": state.total_employment,
        "population": state.total_population,
        "gdp_per_capita": state.gdp_per_capita(annual=True),
        "average_wealth": state.average_wealth,
        "construction_cost": self.model.construction_cost(state),
        "arable_land_consumption": self.model.arable_land_consumption(state),
        "net_infrastructure_usage": float(
            state.building_levels @ self.model.infrastructure_vector
        ),
        "search_success": bool(result.success),
        "search_status": int(result.status),
        "search_message": str(result.message),
    }
    if "nfev" in result:
        values["trial_evaluations"] = int(result.nfev)
    return pd.DataFrame([values])

    def compile(self, scenario: Scenario) -> LinearProblem:
    """Compile *scenario* into an inspectable linear problem."""
    if scenario.objective == "gdp_per_capita":
        raise ValueError(
            "NominalOptimizer does not support the nonlinear "
            "gdp_per_capita objective; use MarketOptimizer."
        )
    if scenario.objective == "average_wealth":
        raise ValueError(
            "NominalOptimizer does not support average_wealth; "
            "use EquilibriumOptimizer."
        )
    data = self._compile_common(scenario)
    if scenario.objective == "gdp":
        objective = -data.gdp_vector
    elif scenario.objective == "employment":
        objective = -self.model.employment_vector
    elif scenario.objective == "automation":
        objective = self.model.employment_vector
    elif scenario.objective == "construction_cost":
        objective = self.model.construction_cost_vector
    else:
        raise ValueError(f"Unknown objective: {scenario.objective!r}")
    problem = self._linear_problem(data, objective)
    self._bind_problem(problem)
    return problem

    def solve_problem(self, problem: LinearProblem) -> EconomyState:
    """Solve a previously compiled linear problem."""
    self._bind_problem(problem)
    result = opt.linprog(
        **problem.linprog_args(),
        bounds=problem.bounds,
        method="highs",
    )
    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")
    self.result = result
    state = self._state_from_levels(problem, np.asarray(result.x, dtype=np.float64))
    self._state = state
    return state

    def import_marginals(self) -> np.ndarray | None:
    """Return objective sensitivities to increasing imports by good.

    Values align to the goods index and measure the minimized LP objective
    change per additional imported unit under the goods-balance floor.
    """
    problem = self.problem
    result = self.result
    if problem is None or result is None:
        return None
    row_slice = problem.inequality_slices.get("min_goods_balance")
    if row_slice is None:
        return None
    ineqlin = getattr(result, "ineqlin", None)
    if ineqlin is None:
        return None
    marginals = getattr(ineqlin, "marginals", None)
    if marginals is None:
        return None
    values = np.asarray(marginals, dtype=np.float64)
    stop = row_slice.stop
    if stop is None or values.shape[0] < stop:
        return None
    return values[row_slice].copy()

    def compile(self, scenario: Scenario) -> MarketProblem:
    """Compile *scenario* into an inspectable nonlinear market problem."""
    if scenario.objective not in ("gdp", "gdp_per_capita"):
        raise ValueError(
            "MarketOptimizer requires objective='gdp' or "
            "objective='gdp_per_capita'."
        )
    data = self._compile_common(scenario)
    warm_start_objective = (
        self.model.employment_vector
        if scenario.objective == "gdp_per_capita" and scenario.goods_export
        else -data.gdp_vector
    )
    warm_start = self._extract_fixed_zero_bounds(
        self._linear_problem(data, warm_start_objective)
    )
    employment = self.model.employment_vector
    economy = self.model

    def objective_terms(levels: np.ndarray) -> tuple[float, np.ndarray]:
        value, gradient = self._value_and_gradient(
            economy,
            levels,
            data.input_matrix,
            data.output_matrix,
            data.net_matrix,
            data.goods_import,
            data.goods_export,
            economy.base_prices,
        )
        if scenario.objective == "gdp_per_capita":
            value, gradient = self._per_capita_value_and_gradient(
                value, gradient, levels, employment
            )
            return (
                value * scenario.workforce_ratio,
                gradient * scenario.workforce_ratio,
            )
        return value, gradient

    def objective(levels: np.ndarray) -> float:
        value, _gradient = objective_terms(levels)
        return -value

    def jacobian(levels: np.ndarray) -> np.ndarray:
        _value, gradient = objective_terms(levels)
        return -gradient

    problem = MarketProblem(
        economy=data.economy,
        scenario=scenario,
        throughput_multipliers=data.throughput_multipliers,
        input_matrix=data.input_matrix,
        output_matrix=data.output_matrix,
        net_matrix=data.net_matrix,
        gdp_vector=data.gdp_vector,
        goods_import=data.goods_import,
        goods_export=data.goods_export,
        A_ub=warm_start.A_ub,
        b_ub=warm_start.b_ub,
        A_eq=warm_start.A_eq,
        b_eq=warm_start.b_eq,
        bounds=warm_start.bounds,
        inequality_slices=warm_start.inequality_slices,
        equality_slices=warm_start.equality_slices,
        objective=objective,
        jacobian=jacobian,
        warm_start=warm_start,
    )
    self._bind_problem(problem)
    return problem

    self._check_feasibility_and_boundedness(warm_start_problem)
    n_buildings = len(self.model.building_index)

    if x0 is None:
        warm_start = opt.linprog(
            **warm_start_problem.linprog_args(),
            bounds=warm_start_problem.bounds,
            method="highs",
        )
        if not warm_start.success:
            raise ValueError(f"Nominal warm-start failed: {warm_start.message}")
        initial = np.asarray(warm_start.x, dtype=np.float64)
        if (
            problem.scenario.objective == "gdp_per_capita"
            and problem.scenario.goods_export
            and float(initial @ self.model.employment_vector) <= 0
        ):
            fallback = replace(
                warm_start_problem, objective_vector=-problem.gdp_vector
            )
            warm_start = opt.linprog(
                **fallback.linprog_args(), bounds=fallback.bounds, method="highs"
            )
            if not warm_start.success:
                raise ValueError(f"Nominal warm-start failed: {warm_start.message}")
            initial = np.asarray(warm_start.x, dtype=np.float64)
    else:
        initial = self._validate_initial_point(
            x0,
            problem.A_ub,
            problem.b_ub,
            problem.A_eq,
            problem.b_eq,
            problem.bounds,
            n_buildings,
        )

    constraints: list[opt.LinearConstraint] = []
    if problem.A_ub is not None and problem.b_ub is not None:
        linear_constraint = cast(Any, opt.LinearConstraint)
        constraints.append(linear_constraint(problem.A_ub, -np.inf, problem.b_ub))
    if problem.A_eq is not None and problem.b_eq is not None:
        linear_constraint = cast(Any, opt.LinearConstraint)
        constraints.append(
            linear_constraint(problem.A_eq, problem.b_eq, problem.b_eq)
        )

    result = opt.minimize(
        problem.objective,
        initial,
        jac=problem.jacobian,
        method="SLSQP",
        bounds=problem.bounds,
        constraints=constraints,
        options=dict(options) if options is not None else None,
    )
    if not result.success:
        raise ValueError(f"Market optimization failed: {result.message}")

    self.result = result
    state = self._state_from_levels(
        problem,
        np.asarray(result.x, dtype=np.float64),
        method="market",
    )
    self._state = state
    return state

    def compile(
    self,
    scenario: Scenario,
    *,
    initial_wealth: np.ndarray | None = None,
    subsidize_operating_losses: bool = True,
    max_iterations: int = 1000,
    rtol: float = 1e-9,
    atol: float = 1e-8,
) -> EquilibriumProblem:
    """Compile shared constraints and freeze the equilibrium evaluation seed.

    Supports ``gdp``, ``gdp_per_capita``, and ``average_wealth``. Household
    consumption is endogenous; imports and exports are fixed external orders.
    Goods-balance floors include trade and exclude household consumption.
    ``subsidize_operating_losses`` funds losses through proportional taxes
    on cooperative and dependent income, without guaranteeing payroll.
    """
    if scenario.objective not in ("gdp", "gdp_per_capita", "average_wealth"):
        raise ValueError(
            "EquilibriumOptimizer requires objective='gdp', "
            "'gdp_per_capita', or 'average_wealth'."
        )
    data = self._compile_common(scenario)
    warm_start = self._extract_fixed_zero_bounds(
        self._linear_problem(data, -data.gdp_vector)
    )
    # Exact zero caps also imply fixed variables. Keep the named inequality
    # rows for inspection; only the search coordinates are reduced.
    if warm_start.A_ub is not None and warm_start.b_ub is not None:
        bounds = list(warm_start.bounds)
        for row, rhs in zip(warm_start.A_ub, warm_start.b_ub, strict=True):
            if rhs == 0 and (row >= 0).all():
                for index in np.flatnonzero(row > 0):
                    bounds[int(index)] = (0.0, 0.0)
        warm_start = replace(warm_start, bounds=tuple(bounds))
    if initial_wealth is None:
        initial_wealth = np.broadcast_to(
            self.model.pop_wealth_init,
            (len(self.model.building_index), len(self.model.pop_index)),
        )
    problem = EquilibriumProblem(
        economy=data.economy,
        scenario=scenario,
        throughput_multipliers=data.throughput_multipliers,
        input_matrix=data.input_matrix,
        output_matrix=data.output_matrix,
        net_matrix=data.net_matrix,
        gdp_vector=data.gdp_vector,
        goods_import=data.goods_import,
        goods_export=data.goods_export,
        A_ub=warm_start.A_ub,
        b_ub=warm_start.b_ub,
        A_eq=warm_start.A_eq,
        b_eq=warm_start.b_eq,
        bounds=warm_start.bounds,
        inequality_slices=warm_start.inequality_slices,
        equality_slices=warm_start.equality_slices,
        warm_start=warm_start,
        initial_wealth=initial_wealth,
        subsidize_operating_losses=subsidize_operating_losses,
        max_iterations=max_iterations,
        rtol=rtol,
        atol=atol,
    )
    self._bind_problem(problem)
    return problem

    self,
    problem: EquilibriumProblem,
    *,
    x0: np.ndarray | None = None,
    options: Mapping[str, object] | None = None,
    log_every: int = 100,

    """Search a compiled problem over a bounded, feasible linear region.

    Return the best feasible equilibrium even on budget exhaustion; inspect
    ``result.success`` and ``result.message`` for search termination.
    ``result.x`` and unscaled ``result.fun`` describe the returned state.
    ``result.solver_result`` retains raw COBYLA output (or ``None`` if it
    was skipped or interrupted by the hard budget). ``nfev`` counts
    distinct consecutive trials, including
    rejected points; ``equilibrium_evaluations`` counts inner solve attempts.
    The initial trial counts against the budget; final verification does not.
    ``equilibrium_failures`` counts failures by status and
    ``last_equilibrium_error`` retains the last failed inner solve.

    Enable ``vic3_analysis.optimize.equilibrium`` at INFO for lifecycle
    events and summaries every ``log_every`` trials (default 100, positive
    integer). DEBUG additionally reports uncached trials and incumbents.
    Logging uses actual maximizing metrics in explicit units, independently
    of native COBYLA ``disp``. Library code never configures log handlers.

    Raises:
        ValueError: For invalid inputs, infeasible/unbounded regions, or no
            feasible converged equilibrium (with diagnostics on ``result``).
        EquilibriumError: If fresh verification of the selected state fails.
    """
    self._bind_problem(problem)
    solver_options, budget = _options(options)
    if (
        isinstance(log_every, (bool, np.bool_))
        or not isinstance(log_every, (int, np.integer))
        or log_every <= 0

        raise ValueError("log_every must be a positive integer.")

        raise

    @staticmethod
def validate_building_levels(
    frame: pd.DataFrame,
    *,
    building_index: tuple[str, ...],
) -> pd.DataFrame:
    """Validate, align, and copy recipes, warning when failed columns are skipped.

    Rows must match ``building_index`` exactly; column order is preserved.
    All-NaN columns are omitted. Each remaining recipe must have finite,
    nonnegative coefficients and at least one positive coefficient. Returned
    float coefficients retain the caller's units without normalization.
    """
    buildings = tuple(building_index)
    rows = tuple(frame.index)
    columns = tuple(frame.columns)
    for name, labels in (
        ("building_index", buildings),
        ("DataFrame index", rows),
        ("DataFrame columns", columns),
    ):
        if any(not isinstance(key, str) or not key for key in labels):
            raise ValueError(f"{name} must contain nonempty string labels.")
        if len(set(labels)) != len(labels):
            raise ValueError(f"{name} must contain unique labels.")
    if not buildings or set(rows) != set(buildings):
        raise ValueError("DataFrame index must match building_index exactly.")
    failed = frame.isna().all(axis=0)
    skipped = [key for key, missing in zip(columns, failed, strict=True) if missing]
    if skipped:
        warnings.warn(
            "Skipping failed supply-chain recipes containing only NaN: "
            + ", ".join(skipped),
            UserWarning,
            stacklevel=2,
        )
    chains = tuple(key for key in columns if key not in skipped)
    if not chains:
        raise ValueError("No usable supply-chain recipes remain.")
    try:
        matrix = frame.loc[list(buildings), list(chains)].to_numpy(
            dtype=np.float64, copy=True
        )
    except (TypeError, ValueError) as exc:
        raise ValueError("matrix must contain numeric coefficients.") from exc
    if not np.isfinite(matrix).all() or (matrix < 0).any():
        raise ValueError("matrix must contain finite non-negative coefficients.")
    if not (matrix > 0).any(axis=0).all():
        raise ValueError("Every chain must have at least one nonzero coefficient.")
    return pd.DataFrame(
        matrix, index=pd.Index(buildings, name="config"), columns=chains
    )

    def compile(
    self,
    scenario: Scenario,
    *,
    initial_wealth: np.ndarray | None = None,
    subsidize_operating_losses: bool = True,
    max_iterations: int = 1000,
    rtol: float = 1e-9,
    atol: float = 1e-8,
) -> SupplyChainEquilibriumProblem:
    """Compile building constraints and the nominal warm start into chains."""
    building = super().compile(
        scenario,
        initial_wealth=initial_wealth,
        subsidize_operating_losses=subsidize_operating_losses,
        max_iterations=max_iterations,
        rtol=rtol,
        atol=atol,
    )
    matrix = self._chain_matrix
    A_ub = None if building.A_ub is None else building.A_ub @ matrix
    A_eq = None if building.A_eq is None else building.A_eq @ matrix
    b_ub = building.b_ub
    slices = dict(building.inequality_slices)
    bounds: Bounds = tuple((0.0, None) for _ in self.chain_index)
    blocked = np.zeros(len(bounds), dtype=bool)
    rows: list[np.ndarray] = []
    rhs: list[float] = []
    for row, (lower, upper) in zip(matrix, building.bounds, strict=True):
        if upper == 0:
            blocked |= row > 0
        elif upper is not None:
            rows.append(row)
            rhs.append(upper)
        if lower > 0:
            rows.append(-row)
            rhs.append(-lower)
    if rows:
        start = 0 if b_ub is None else len(b_ub)
        extra = np.stack(rows)
        A_ub = extra if A_ub is None else np.vstack((A_ub, extra))
        b_ub = np.asarray(rhs) if b_ub is None else np.concatenate((b_ub, rhs))
        slices["building_bounds"] = slice(start, start + len(rhs))
    bounds = tuple((0.0, 0.0 if used else None) for used in blocked)
    warm = self._extract_fixed_zero_bounds(
        replace(
            building.warm_start,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            bounds=bounds,
            inequality_slices=slices,
            objective_vector=matrix.T @ building.warm_start.objective_vector,
        )
    )
    problem = SupplyChainEquilibriumProblem(
        economy=building.economy,
        scenario=building.scenario,
        throughput_multipliers=building.throughput_multipliers,
        input_matrix=building.input_matrix,
        output_matrix=building.output_matrix,
        net_matrix=building.net_matrix,
        gdp_vector=building.gdp_vector,
        goods_import=building.goods_import,
        goods_export=building.goods_export,
        A_ub=warm.A_ub,
        b_ub=warm.b_ub,
        A_eq=warm.A_eq,
        b_eq=warm.b_eq,
        bounds=warm.bounds,
        inequality_slices=warm.inequality_slices,
        equality_slices=warm.equality_slices,
        warm_start=warm,
        initial_wealth=building.initial_wealth,
        subsidize_operating_losses=building.subsidize_operating_losses,
        max_iterations=building.max_iterations,
        rtol=building.rtol,
        atol=building.atol,
        chain_index=self.chain_index,
        chain_matrix=matrix,
        _owner=self._owner,
        building_problem=building,
    )
    self._bind_problem(problem)
    return problem

    def solve_problem(
    self,
    problem: EquilibriumProblem,
    *,
    x0: np.ndarray | None = None,
    options: Mapping[str, object] | None = None,
    log_every: int = 100,
) -> EconomyState:
    """Search chain coordinates and verify the original building constraints."""
    if not isinstance(problem, SupplyChainEquilibriumProblem):
        raise TypeError("Expected a SupplyChainEquilibriumProblem.")
    if (
        problem._owner is not self._owner
        or problem.chain_index != self.chain_index
        or not np.array_equal(problem.chain_matrix, self._chain_matrix)
    ):
        raise ValueError(
            "Compiled problem belongs to a different supply-chain optimizer or recipes."
        )
    try:
        state = super().solve_problem(
            problem, x0=x0, options=options, log_every=log_every
        )
    finally:
        if self.result is not None:
            self.result.building_levels = (
                None
                if self.result.x is None
                else problem.to_building_levels(self.result.x)
            )
    if not _is_feasible(problem.building_problem, state.building_levels):
        self._state = None
        if self.result is not None:
            self.result.success = False
            self.result.message = (
                "Supply-chain final building feasibility check failed."
            )
        raise ValueError("Supply-chain final building feasibility check failed.")
    return state
